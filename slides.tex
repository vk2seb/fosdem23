\begin{frame}{\texttt{tiliqua} DSP library - some components}
    \begin{itemize}
        \item \textbf{Resample} -- Rational resampler.
        \item \textbf{PitchShift} -- Granular pitch shifter.
        \item \textbf{DelayLine} -- Memory-backed audio delay line.
        \item \textbf{Waveshaper} -- Interpolating waveshaper.
        \item \textbf{FIR} -- FIR filter.
        \item \textbf{VCA} -- Voltage controlled amplifier.
        \item \textbf{NCO} -- Numerically controlled oscillator.
        \item \textbf{Trigger} -- Stream hysteresis.
        \item \textbf{Ramp} -- Synchronized ramp generator.
        \item \textbf{SVF} -- Tunable state-variable filter (HPF/BPF/LPF).
        \item \textbf{MatrixMix} -- Matrix mixer with tweakable coefficients.
        \item \textbf{Boxcar} -- Cheap averaging without multipliers.
    \end{itemize}
\end{frame}

\begin{frame}{\texttt{tiliqua} DSP library - frequency-domain processing (new!)}
    \begin{itemize}
        \item \textbf{FFT} -- forward + inverse, iterative, low area usage.
        \item \textbf{CORDIC} -- complex/polar conversions.
        \item \textbf{Window} -- windowing core and selectable window functions.
        \item \textbf{STFT} -- forward + inverse, frequency-domain processing of time-domain signals.
        \item \textbf{SpectralEnvelope} -- compute filtered power spectra over time.
        \item \textbf{SpectralCrossSynthesis} -- vocoder-like effect.
    \end{itemize}
    \begin{block}{Also: check out Amaranth \texttt{lib.fixed}}

\begin{frame}{Dynamic clocking (new!)}
    \begin{itemize}
        \item People have different screens (pixel clock).
        \item Same bitstream should work everywhere.

\begin{frame}{USB2 Host (new!)}

        \item Solution: pure-gateware USB2 high-speed host. (first?)
        \item Details: \texttt{https://github.com/apfaudio/tiliqua/pull/65}
