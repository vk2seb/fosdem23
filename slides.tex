\begin{frame}{New things!}
    \begin{itemize}
        \item \textbf{More DSP cores, lib.fixed} -- FFTs, STFTs, frequency-domain stuff!
        \item \textbf{Dynamic pixel clocks} -- hot-swap screens.
        \item \textbf{USB2 Host} -- in pure gateware, based on LUNA.
        \item \textbf{Persistent options} -- bitstream auto-restore on power loss.
        \item \textbf{Tiny GPU} -- hardware-accelerated fonts/line drawing.
    \end{itemize}
\end{frame}

\begin{frame}{\texttt{tiliqua} DSP library - frequency-domain processing (new!)}
    \begin{itemize}
        \item \textbf{FFT} -- forward + inverse, iterative, low area usage.
        \item \textbf{CORDIC} -- complex/polar conversions.
        \item \textbf{Window} -- windowing core and selectable window functions.
        \item \textbf{STFT} -- forward + inverse, frequency-domain processing of time-domain signals.
        \item \textbf{SpectralEnvelope} -- compute filtered power spectra over time.
        \item \textbf{SpectralCrossSynthesis} -- vocoder-like effect.
    \end{itemize}
    \begin{block}{Also: check out Amaranth \texttt{lib.fixed}}
        \begin{itemize}
            \item \textbf{Extension to Amaranth standard library}
            \item \textbf{RFC + Implementation (WIP)}: github.com/amaranth-lang/amaranth/pull/1578
            \item \textbf{Comments welcome!}
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Dynamic clocking (new!)}
    \begin{center}
        \includegraphics[height=5.1cm]{img3/scrn_back.jpg}
    \end{center}
    \begin{itemize}
        \item People have different screens (pixel clock).
        \item Same bitstream should work everywhere.
        \item Core reads display EDID, dynamically adapts resolution!
        \item Details: \texttt{github.com/apfaudio/tiliqua/pull/107}
    \end{itemize}
\end{frame}

\begin{frame}{USB2 Host (new!)}
    \begin{center}
        \includegraphics[height=5.1cm]{img3/usb_midi.jpg}
    \end{center}
    \begin{itemize}
        \item Cheapest MIDI keyboards only have USB MIDI :(
        \item Solution: pure-gateware USB2 high-speed host. (first?)
        \item Details: \texttt{github.com/apfaudio/tiliqua/pull/65}
        \item Builds on LUNA: \texttt{github.com/greatscottgadgets/luna}
    \end{itemize}
\end{frame}

\begin{frame}{Persistent Options (new!)}
    \begin{center}
        \includegraphics[height=5cm]{img3/autoboot.jpg}
    \end{center}
    \begin{itemize}
        \item Wear-levelled persistent storage to configuration flash
        \item Store last bitstream, all options, autoboot.
        \item Details: \texttt{github.com/apfaudio/tiliqua/pull/129}
    \end{itemize}
\end{frame}
